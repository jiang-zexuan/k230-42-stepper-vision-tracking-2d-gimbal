# PanView T07 独立学习文档

> 生成日期：2026-09-04  
> 学习目标：脱离 STM32 完整工程，理解“运动命令如何安全地驱动双轴步进电机”。

本文只保留 T07 的核心逻辑：读取运动命令、速度渐变、方向控制、脉冲位置累计、软件限位、人工零点和启停。真实工程中的 CubeMX 初始化、FreeRTOS 任务、消息队列、串口日志等内容不放在这里。

信息流：

```text
MotionCommand
    ↓
速度斜坡
    ↓
双轴限速与软件限位
    ↓
DIR 设置 + STEP 脉冲
    ↓
理论位置累计
```

## 1. 任务需求说明

### 1.1 T07 要解决什么问题

T06 已经能够根据视觉误差生成 `MotionCommand`，但那只是一个数字，例如“水平轴向正方向以 500 step/s 转动”。T07 要把这条命令真正交给两个步进电机执行，同时保证：

1. 两个轴方向正确。
2. 速度不会瞬间跳变。
3. 速度不会超过上限。
4. 位置不会超过软件限制。
5. 复位后不会直接乱转，必须人工确认零点。
6. 运行中按键可以暂停并失能电机。

### 1.2 输入

```c
typedef struct
{
  int32_t pan_speed_steps_per_second;   /* 水平速度，正负表示方向 */
  int32_t pitch_speed_steps_per_second; /* 俯仰速度，正负表示方向 */
  uint8_t valid;                        /* 1=命令有效，0=停止 */
} MotionCommand;
```

输入来自 MotionTask。T07 只读取输入，不修改输入。

### 1.3 输出

- 水平轴：DIR 电平、STEP 定时器脉冲、理论位置。
- 俯仰轴：DIR 电平、STEP 定时器脉冲、理论位置。
- 可读状态：当前速度、当前位置、是否运行、系统是否已确认零点。

### 1.4 依赖的硬件/底层模块

- STM32F407 GPIO：输出 `DIR` 和 `EN`。
- STM32F407 TIM3/TIM4：输出 STEP 脉冲。
- 定时器更新中断：每个脉冲完成后累计理论位置。
- 一个按键：确认零点、开始和暂停。
- X42S 步进电机驱动器及独立电机电源。

当前工程中 TIM3/TIM4 的实际定时器时钟是 84 MHz。若错误使用 16 MHz 计算，实际速度会变成命令速度的 `84/16=5.25` 倍。

## 2. 解耦后的最小可阅读代码

下面代码使用简单的硬件抽象接口。真实 STM32 工程只需要把这些接口连接到 HAL GPIO 和定时器函数即可。

### 2.1 `t07_stepper.h`

```c
#ifndef T07_STEPPER_H
#define T07_STEPPER_H

#include <stdint.h>

/* 上层传给执行模块的双轴速度命令。速度单位：step/s。 */
typedef struct
{
  int32_t pan_speed_steps_per_second;
  int32_t pitch_speed_steps_per_second;
  uint8_t valid;
} MotionCommand;

/* 执行模块向上层提供的只读状态。 */
typedef struct
{
  int32_t pan_position_steps;
  int32_t pitch_position_steps;
  int32_t pan_applied_speed_steps_per_second;
  int32_t pitch_applied_speed_steps_per_second;
  uint8_t pan_running;
  uint8_t pitch_running;
  uint8_t system_running;
} T07StepperStatus;

/* 真实工程中由 HAL 或板级代码实现这 6 个硬件动作。 */
typedef struct
{
  void (*set_pan_direction)(int8_t direction);
  void (*set_pitch_direction)(int8_t direction);
  void (*set_pan_enable)(uint8_t enabled);
  void (*set_pitch_enable)(uint8_t enabled);
  void (*set_pan_step_rate)(uint32_t steps_per_second);
  void (*set_pitch_step_rate)(uint32_t steps_per_second);
} T07StepperHardware;

void T07_Stepper_Init(const T07StepperHardware *hardware);
void T07_Stepper_ToggleRun(void);
void T07_Stepper_Execute(const MotionCommand *command);
void T07_Stepper_OnPanStep(void);
void T07_Stepper_OnPitchStep(void);
void T07_Stepper_GetStatus(T07StepperStatus *status);

#endif
```

### 2.2 `t07_stepper.c`

```c
#include "t07_stepper.h"

#define T07_PAN_LIMIT_STEPS        800
#define T07_PITCH_LIMIT_STEPS      800
#define T07_PAN_MAX_SPEED          820
#define T07_PITCH_MAX_SPEED        800
#define T07_SPEED_CHANGE_PER_CALL  60

static T07StepperHardware hw;
static int32_t pan_position;
static int32_t pitch_position;
static int32_t pan_speed;
static int32_t pitch_speed;
static int8_t pan_direction;
static int8_t pitch_direction;
static uint8_t pan_running;
static uint8_t pitch_running;
static uint8_t system_running;

/* 把速度限制在 [-maximum, +maximum]。 */
static int32_t limit_speed(int32_t speed, int32_t maximum)
{
  if (speed > maximum) return maximum;
  if (speed < -maximum) return -maximum;
  return speed;
}

/* 每次只改变一小步，避免电机速度突然变化。 */
static int32_t ramp_speed(int32_t current, int32_t target)
{
  int32_t difference = target - current;

  if (difference > T07_SPEED_CHANGE_PER_CALL)
    return current + T07_SPEED_CHANGE_PER_CALL;
  if (difference < -T07_SPEED_CHANGE_PER_CALL)
    return current - T07_SPEED_CHANGE_PER_CALL;
  return target;
}

static void stop_all(void)
{
  hw.set_pan_step_rate(0);
  hw.set_pitch_step_rate(0);
  hw.set_pan_enable(system_running);
  hw.set_pitch_enable(system_running);
  pan_running = 0;
  pitch_running = 0;
  pan_speed = 0;
  pitch_speed = 0;
}

void T07_Stepper_Init(const T07StepperHardware *hardware)
{
  hw = *hardware;
  pan_position = 0;
  pitch_position = 0;
  pan_speed = 0;
  pitch_speed = 0;
  pan_direction = 0;
  pitch_direction = 0;
  pan_running = 0;
  pitch_running = 0;
  system_running = 0;

  /* 复位后先失能，等待操作者把机械结构摆到中心。 */
  hw.set_pan_enable(0);
  hw.set_pitch_enable(0);
}

void T07_Stepper_ToggleRun(void)
{
  if (system_running)
  {
    /* 运行中按键：暂停并失能。 */
    system_running = 0;
    stop_all();
    return;
  }

  /* 开始运行时，当前位置被定义为新的软件零点。 */
  pan_position = 0;
  pitch_position = 0;
  pan_speed = 0;
  pitch_speed = 0;
  system_running = 1;
  hw.set_pan_enable(1);
  hw.set_pitch_enable(1);
}

void T07_Stepper_Execute(const MotionCommand *command)
{
  int32_t pan_target = 0;
  int32_t pitch_target = 0;

  if (!system_running)
  {
    stop_all();
    return;
  }

  if (command != 0 && command->valid)
  {
    pan_target = limit_speed(command->pan_speed_steps_per_second,
                             T07_PAN_MAX_SPEED);
    pitch_target = limit_speed(command->pitch_speed_steps_per_second,
                               T07_PITCH_MAX_SPEED);
  }

  pan_speed = ramp_speed(pan_speed, pan_target);
  pitch_speed = ramp_speed(pitch_speed, pitch_target);

  /* 到达软件限位后，禁止继续向外运动。 */
  if ((pan_position >= T07_PAN_LIMIT_STEPS && pan_speed > 0) ||
      (pan_position <= -T07_PAN_LIMIT_STEPS && pan_speed < 0))
    pan_speed = 0;
  if ((pitch_position >= T07_PITCH_LIMIT_STEPS && pitch_speed > 0) ||
      (pitch_position <= -T07_PITCH_LIMIT_STEPS && pitch_speed < 0))
    pitch_speed = 0;

  pan_direction = (pan_speed > 0) - (pan_speed < 0);
  pitch_direction = (pitch_speed > 0) - (pitch_speed < 0);

  hw.set_pan_direction(pan_direction);
  hw.set_pitch_direction(pitch_direction);
  hw.set_pan_step_rate((uint32_t)(pan_speed < 0 ? -pan_speed : pan_speed));
  hw.set_pitch_step_rate((uint32_t)(pitch_speed < 0 ? -pitch_speed : pitch_speed));
  hw.set_pan_enable(1);
  hw.set_pitch_enable(1);
  pan_running = pan_speed != 0;
  pitch_running = pitch_speed != 0;
}

void T07_Stepper_OnPanStep(void)
{
  if (pan_running)
    pan_position += pan_direction;
}

void T07_Stepper_OnPitchStep(void)
{
  if (pitch_running)
    pitch_position += pitch_direction;
}

void T07_Stepper_GetStatus(T07StepperStatus *status)
{
  if (status == 0) return;
  status->pan_position_steps = pan_position;
  status->pitch_position_steps = pitch_position;
  status->pan_applied_speed_steps_per_second = pan_speed;
  status->pitch_applied_speed_steps_per_second = pitch_speed;
  status->pan_running = pan_running;
  status->pitch_running = pitch_running;
  status->system_running = system_running;
}
```

#include "t07_speed_ramp.h"

int32_t T07_SpeedRamp_Step(int32_t current, int32_t target,
                           int32_t max_change)
{
  int32_t difference = target - current;
  if (difference > max_change) return current + max_change;
  if (difference < -max_change) return current - max_change;
  return target;
}

## 3. 代码拆解讲解

### 3.1 变量和结构体成员

- `hw`：保存硬件操作函数，解耦算法和 HAL。
- `pan_position`、`pitch_position`：理论脉冲位置；不是编码器真实位置。
- `pan_speed`、`pitch_speed`：经过速度斜坡后的实际执行速度。
- `pan_direction`、`pitch_direction`：方向，`1` 正向、`-1` 反向、`0` 停止。
- `pan_running`、`pitch_running`：对应轴当前是否有 STEP 输出。
- `system_running`：是否已经确认零点并允许运行。
- `MotionCommand.valid`：命令是否有效；无效命令按停止处理。

### 3.2 对外接口

- `T07_Stepper_Init()`：上电初始化，传入硬件函数表。
- `T07_Stepper_ToggleRun()`：模拟 KEY_UP。暂停状态开始并清零软件位置；运行状态暂停并失能。
- `T07_Stepper_Execute()`：周期调用，输入最新命令，完成限速、斜坡、限位和硬件输出。
- `T07_Stepper_OnPanStep()`：水平定时器产生一个 STEP 后调用。
- `T07_Stepper_OnPitchStep()`：俯仰定时器产生一个 STEP 后调用。
- `T07_Stepper_GetStatus()`：读取状态用于串口日志、UI 或调试器观察。
- `T07_SpeedRamp_Step()`：独立的通用速度渐变函数。

### 3.3 私有函数

- `limit_speed()`：限制最大速度。
- `ramp_speed()`：让当前速度逐步靠近目标速度。
- `stop_all()`：停止两个轴的 STEP，并根据系统状态决定是否保持使能。

### 3.4 完整执行流程

1. 复位后调用 `Init`，两个电机失能。
2. 操作者手动把云台摆到机械中心。
3. 第一次按键调用 `ToggleRun`，清零软件位置并使能电机。
4. 周期任务读取最新 `MotionCommand`，调用 `Execute`。
5. `Execute` 先检查系统是否运行，再限制速度。
6. 速度斜坡限制本次最多变化 60 step/s。
7. 如果继续向软件限位外运动，则把该轴速度置零。
8. 设置方向、脉冲频率和使能状态。
9. 定时器每产生一个 STEP，就调用对应的 `On...Step` 累计理论位置。
10. 再次按键时停止脉冲、清零执行速度并失能。

### 3.5 嵌入式开发坑点

1. **定时器时钟必须核对**：不能凭印象使用 16 MHz；当前 TIM3/TIM4 使用 84 MHz。
2. **理论位置不是绝对位置**：丢步、堵转或手动转动后必须重新回中。
3. **STEP 停止和 EN 失能不同**：运行中无目标时通常停脉冲但保持力矩；暂停时才失能。
4. **方向切换要先减速**：不能让定时器在正转时直接切成反转。
5. **限位要在输出前检查**：只在日志里提示限位没有保护作用。
6. **中断共享变量要注意并发**：真实工程中的位置和运行标志应使用 `volatile`，并避免在中断和任务中同时做复杂修改。
7. **无效命令要有确定行为**：本模块把 `valid=0` 定义为停止，而不是继续执行上一条命令。

## 4. `main` 简易测试 Demo

下面 Demo 不需要 STM32，使用打印语句模拟硬件动作，重点观察调用关系。

```c
#include <stdio.h>
#include "t07_stepper.h"

static void set_pan_direction(int8_t d)  { printf("pan DIR=%d\n", d); }
static void set_pitch_direction(int8_t d){ printf("pitch DIR=%d\n", d); }
static void set_pan_enable(uint8_t e)    { printf("pan EN=%d\n", e); }
static void set_pitch_enable(uint8_t e)  { printf("pitch EN=%d\n", e); }
static void set_pan_rate(uint32_t v)     { printf("pan STEP=%lu\n", (unsigned long)v); }
static void set_pitch_rate(uint32_t v)   { printf("pitch STEP=%lu\n", (unsigned long)v); }

int main(void)
{
  T07StepperHardware hw = {
    set_pan_direction, set_pitch_direction,
    set_pan_enable, set_pitch_enable,
    set_pan_rate, set_pitch_rate
  };
  T07StepperStatus status;
  MotionCommand command = { 600, -300, 1 };

  T07_Stepper_Init(&hw);       /* 复位：电机失能 */
  T07_Stepper_ToggleRun();     /* 按 KEY_UP：确认零点并开始 */

  for (int i = 0; i < 12; ++i)
  {
    T07_Stepper_Execute(&command); /* 模拟周期任务 */
    T07_Stepper_OnPanStep();       /* 模拟水平定时器 STEP */
    T07_Stepper_OnPitchStep();     /* 模拟俯仰定时器 STEP */
  }

  T07_Stepper_GetStatus(&status);
  printf("pan_pos=%ld pitch_pos=%ld pan_v=%ld pitch_v=%ld\n",
         (long)status.pan_position_steps,
         (long)status.pitch_position_steps,
         (long)status.pan_applied_speed_steps_per_second,
         (long)status.pitch_applied_speed_steps_per_second);

  T07_Stepper_ToggleRun();     /* 再按 KEY_UP：暂停并失能 */
  return 0;
}
```

这个 Demo 的关键观察点是：第一次执行时速度不是立即变成 600，而是从 0 逐步增加；调用 `OnPanStep` 和 `OnPitchStep` 后位置才累计；最后再次按键会停止并失能。
